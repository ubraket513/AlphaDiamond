cmake_minimum_required(VERSION 3.30)
project(AlphaDiamondDesktop LANGUAGES CXX)

if(MSVC)
    set(CMAKE_CXX_STANDARD 23)
    add_compile_options(/std:c++latest)
else()
    set(CMAKE_CXX_STANDARD 26)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(CMAKE_CONFIGURATION_TYPES OR NOT CMAKE_BUILD_TYPE)
    message(FATAL_ERROR "Desktop requires a single explicit CMAKE_BUILD_TYPE matching the SDK lock")
endif()
option(DIAMOND_BUILD_QT_SOO "Enable model inference" ON)
option(DIAMOND_DESKTOP_ENABLE_RATINGS "Enable local ratings" ON)
set(DIAMOND_ENGINE_SDK_PREFIX "" CACHE PATH "Installed AlphaDiamond Engine SDK prefix")
set(DIAMOND_ENGINE_SDK_LOCK "${CMAKE_CURRENT_SOURCE_DIR}/engine-sdk.lock.json" CACHE FILEPATH "SDK lock file")
set(DIAMOND_ENGINE_SDK_ARCHIVE "" CACHE FILEPATH "Optional SDK archive to verify against lock")
if(NOT DIAMOND_ENGINE_SDK_PREFIX)
    message(FATAL_ERROR "Set DIAMOND_ENGINE_SDK_PREFIX to the installed Engine SDK")
endif()
find_package(AlphaDiamondEngine 0.1.0 EXACT CONFIG REQUIRED COMPONENTS Runtime
    PATHS "${DIAMOND_ENGINE_SDK_PREFIX}" NO_DEFAULT_PATH)
include(cmake/VerifyEngineSdk.cmake)
verify_engine_sdk("${DIAMOND_ENGINE_SDK_LOCK}" "${AlphaDiamondEngine_MANIFEST}" "${DIAMOND_ENGINE_SDK_PREFIX}" "${DIAMOND_ENGINE_SDK_ARCHIVE}")
set(DIAMOND_DESKTOP_STANDALONE ON)
if(BUILD_TESTING AND DIAMOND_BUILD_QT_SOO)
    if(NOT DIAMOND_QT_MIN_TEST_ARTIFACT)
        set(DIAMOND_QT_MIN_TEST_ARTIFACT "${AlphaDiamondEngine_DATA_DIR}/fixtures/min" CACHE PATH "Min deployment artifact for Qt inference test" FORCE)
    endif()
    if(NOT EXISTS "${DIAMOND_QT_MIN_TEST_ARTIFACT}/metadata.json")
        message(FATAL_ERROR "Desktop inference tests require the SDK Min fixture")
    endif()
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/qt" "${CMAKE_CURRENT_BINARY_DIR}/app")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/resources/model-sources.json" model_sources)
string(JSON github_repository GET "${model_sources}" github_repository)
string(JSON huggingface_bucket GET "${model_sources}" huggingface_bucket)
foreach(identity IN ITEMS github_repository huggingface_bucket)
    if(NOT "${${identity}}" MATCHES "^[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$")
        message(FATAL_ERROR "Invalid model source identity: ${identity}")
    endif()
endforeach()
target_compile_definitions(diamond_qt_backend PRIVATE
    "DIAMOND_MODEL_GITHUB_REPOSITORY=\"${github_repository}\""
    "DIAMOND_MODEL_HUGGINGFACE_BUCKET=\"${huggingface_bucket}\"")
